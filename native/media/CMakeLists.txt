cmake_minimum_required(VERSION 3.22)
project(pinwheel_media LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Pinwheel media requires Windows x64")
endif()
if(NOT EXISTS "$ENV{JAVA_HOME}/include/jni.h")
  message(FATAL_ERROR "JAVA_HOME must name a JDK")
endif()
add_library(pinwheel_media SHARED media_bridge.cpp)
target_include_directories(pinwheel_media PRIVATE "$ENV{JAVA_HOME}/include" "$ENV{JAVA_HOME}/include/win32" "${FFMPEG_SDK}/include")
target_link_libraries(pinwheel_media PRIVATE "${FFMPEG_SDK}/lib/avformat.lib" "${FFMPEG_SDK}/lib/avcodec.lib" "${FFMPEG_SDK}/lib/avutil.lib" "${FFMPEG_SDK}/lib/swscale.lib" "${FFMPEG_SDK}/lib/swresample.lib")
target_compile_definitions(pinwheel_media PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(pinwheel_media PRIVATE /W4 /EHsc)
set_target_properties(pinwheel_media PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${NATIVE_OUTPUT}")
